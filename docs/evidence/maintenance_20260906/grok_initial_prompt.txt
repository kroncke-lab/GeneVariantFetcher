21: def extract_pmids_from_html(html: Union[str, BeautifulSoup]) -> Set[str]:
22:     """
23:     Extract PubMed IDs (PMIDs) from HTML content.
24:
25:     This function uses multiple strategies to find PMIDs:
26:     1. Links to pubmed.ncbi.nlm.nih.gov with numeric IDs
27:     2. Text patterns like "PMID: 12345678" or "PMID:12345678"
28:     3. Standalone 7-10 digit numbers in relevant contexts
29:     4. Table cells and list items containing PMIDs
30:
31:     Args:
32:         html: Either raw HTML string or a BeautifulSoup object
33:
34:     Returns:
35:         Set of PMID strings (numeric IDs as strings)
36:
37:     Example:
38:         >>> html = '<a href="https://pubmed.ncbi.nlm.nih.gov/12345678">Paper</a>'
39:         >>> pmids = extract_pmids_from_html(html)
40:         >>> print(pmids)
41:         {'12345678'}
42:     """
43:     warn_deprecated(
44:         "utils.html_utils.extract_pmids_from_html",
45:         "a source-specific parser (PubMind keeps its parser private)",
46:     )
47:
48:     # Convert to BeautifulSoup if necessary
49:     if isinstance(html, str):
50:         soup = BeautifulSoup(html, "html.parser")
51:     else:
52:         soup = html
53:
54:     pmids = set()
55:
56:     # Strategy 1: Extract from PubMed links
57:     # Look for URLs like https://pubmed.ncbi.nlm.nih.gov/12345678
58:     pubmed_links = soup.find_all(
59:         "a", href=re.compile(r"pubmed\.ncbi\.nlm\.nih\.gov/(\d+)")
60:     )
61:     for link in pubmed_links:
62:         match = re.search(r"pubmed\.ncbi\.nlm\.nih\.gov/(\d+)", link.get("href", ""))
63:         if match:
64:             pmid = match.group(1)
65:             if _is_valid_pmid(pmid):
66:                 pmids.add(pmid)
67:                 logger.debug(f"Found PMID {pmid} in link")
68:
69:     # Strategy 2: Extract from "PMID:" patterns in text
70:     # Matches patterns like "PMID: 12345678" or "PMID:12345678"
71:     pmid_pattern = re.compile(r"PMID:?\s*(\d{7,10})", re.IGNORECASE)
72:
73:     # Search in all text content
74:     all_text = soup.get_text()
75:     for match in pmid_pattern.finditer(all_text):
76:         pmid = match.group(1)
77:         if _is_valid_pmid(pmid):
78:             pmids.add(pmid)
79:             logger.debug(f"Found PMID {pmid} in text pattern")
80:
81:     # Strategy 3: Extract from table cells (common in PubMind results)
82:     # Look for table cells that might contain PMIDs
83:     for cell in soup.find_all(["td", "th"]):
84:         cell_text = cell.get_text().strip()
85:         # Look for standalone numbers that could be PMIDs
86:         numbers = re.findall(r"(?<!\d)(\d{7,10})(?!\d)", cell_text)
87:         for num in numbers:
88:             if _is_valid_pmid(num):
89:                 # Additional validation: check if context suggests it's a PMID
90:                 context = cell_text.lower()
91:                 if any(
92:                     keyword in context
93:                     for keyword in ["pmid", "pubmed", "article", "paper"]
94:                 ):
95:                     pmids.add(num)
96:                     logger.debug(f"Found PMID {num} in table cell")
97:
98:     # Strategy 4: Extract from anchor text and titles
99:     for link in soup.find_all("a"):
100:         link_text = link.get_text()
101:         for match in pmid_pattern.finditer(link_text):
102:             pmid = match.group(1)
103:             if _is_valid_pmid(pmid):
104:                 pmids.add(pmid)
105:                 logger.debug(f"Found PMID {pmid} in link text")
106:
107:     logger.info(f"Extracted {len(pmids)} unique PMIDs from HTML")
108:     return pmids

111: def _is_valid_pmid(pmid: str) -> bool:
112:     """
113:     Validate that a string is a plausible PMID.
114:
115:     PMIDs are typically 7-10 digit numbers. This function performs
116:     basic validation to filter out obvious non-PMIDs.
117:
118:     Args:
119:         pmid: String to validate
120:
121:     Returns:
122:         True if the string could be a valid PMID
123:     """
124:     if not pmid or not pmid.isdigit():
125:         return False
126:
127:     # PMIDs are typically 7-10 digits
128:     if len(pmid) < 7 or len(pmid) > 10:
129:         return False
130:
131:     # PMIDs don't start with 0
132:     if pmid.startswith("0"):
133:         return False
134:
135:     return True
