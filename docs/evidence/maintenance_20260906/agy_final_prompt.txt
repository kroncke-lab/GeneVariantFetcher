### benchmarks/evaluation_tiers/export_cost_usage_receipts.py
1: #!/usr/bin/env python3
2: """Export usage-only calibration evidence for tests on a fresh checkout.
3:
4: Requires the original ignored operator traces. Does not change the SHA-pinned
5: cost profile or registries, and never exports prompts or response text.
6: """
7:
8: import argparse
9: import hashlib
10: import json
11: from pathlib import Path
12:
13: ROOT = Path(__file__).resolve().parents[2]
14:
15:
16: def sha(path: Path) -> str:
17:     return hashlib.sha256(path.read_bytes()).hexdigest()
18:
19:
20: def export(profile_path: Path, destination: Path) -> dict:
21:     profile = json.loads(profile_path.read_text())
22:     result = {
23:         "schema_version": 1,
24:         "purpose": "Portable usage-only receipts from local calibration traces; no prompts, responses, credentials or paper text. Original calibration/profile bytes remain unchanged.",
25:         "calibration_sha256": sha(profile_path),
26:         "calibrations": {},
27:     }
28:     for gene, calibration in profile["calibrations"].items():
29:         source = ROOT / calibration["source"]
30:         if source.name == "predictions.json":
31:             continue  # These complete locked predictions are already tracked.
32:         if sha(source) != calibration["source_sha256"]:
33:             raise ValueError(f"Calibration source changed: {source}")
34:         calls = []
35:         for path in sorted(source.parent.rglob("*.json")):
36:             if path == source:
37:                 continue
38:             try:
39:                 record = json.loads(path.read_text())
40:             except (OSError, ValueError):
41:                 continue
42:             if record.get("record_type") != "llm_call":
43:                 continue
44:             usage = (record.get("response") or {}).get("usage") or {}
45:             input_tokens = int(
46:                 usage.get("prompt_tokens") or usage.get("input_tokens") or 0
47:             )
48:             total_tokens = int(usage.get("total_tokens") or 0)
49:             calls.append(
50:                 {
51:                     "trace_sha256": sha(path),
52:                     "model": str(
53:                         (record.get("context") or {}).get("model") or "unknown"
54:                     ),
55:                     "input_tokens": input_tokens,
56:                     "output_tokens": max(total_tokens - input_tokens, 0),
57:                 }
58:             )
59:         actual = {}
60:         for call in calls:
61:             counts = actual.setdefault(
62:                 call["model"], dict(calls=0, input_tokens=0, output_tokens=0)
63:             )
64:             counts["calls"] += 1
65:             counts["input_tokens"] += call["input_tokens"]
66:             counts["output_tokens"] += call["output_tokens"]
67:         if actual != calibration["models"]:
68:             raise ValueError(f"Trace usage disagrees with calibration: {gene}")
69:         result["calibrations"][gene] = {
70:             "source": calibration["source"],
71:             "source_sha256": calibration["source_sha256"],
72:             "attempts": calibration["attempts"],
73:             "calls": calls,
74:         }
75:     destination.parent.mkdir(parents=True, exist_ok=True)
76:     destination.write_text(json.dumps(result, indent=2) + "\n")
77:     return result
78:
79:
80: def main():
81:     parser = argparse.ArgumentParser(description=__doc__)
82:     parser.add_argument(
83:         "--profile",
84:         type=Path,
85:         default=Path(__file__).with_name("cost_calibration.json"),
86:     )
87:     parser.add_argument("--out", type=Path, required=True)
88:     args = parser.parse_args()
89:     result = export(args.profile, args.out)
90:     print(
91:         f"Exported {len(result['calibrations'])} usage receipts; SHA-256 {sha(args.out)}"
92:     )
93:
94:
95: if __name__ == "__main__":
96:     main()

257: paywall/cookie recovery pass, use:
258:
259: ```bash
260: # Works without publisher keys; skips live paywall recovery
261: gvf gvf-run KCNH2 --email brett.kroncke@gmail.com --output ./results --no-source-recovery
262: ```
263:
264: `--no-source-recovery` is not a strict PMC-only switch: normal harvesting can
265: still use cached source and its configured free-text/publisher routes. It skips
266: the later recovery pass, so inaccessible bodies or missing supplements can
267: remain unresolved. Publisher coverage and credentials are described in
268: [API_KEYS.md](API_KEYS.md).

270: ## Troubleshooting
271:
272: ### "No PMIDs found"
273: - Check your internet connection
274: - Verify the gene symbol is correct
275: - Some rare genes have limited literature
276:
277: ### "LLM API error"
278: - Verify the selected provider's credential is present without printing it;
279:   use the configuration/status checks in [TESTING.md](TESTING.md).
280: - Check the selected provider account has credits or quota
281: - Rate limits may require waiting
282:
283: ### "Few papers downloaded"
284: - Inspect whether the source is absent, abstract-only, or missing a required supplement
285: - Add publisher API keys for better coverage
286: - Check `pmc_fulltext/paywalled_missing.csv` for inaccessible papers
287:
288: ### Extraction seems slow
289: - LLM calls take time; this is normal
290: - For a bounded fetch test, run `gvf extract ... --output ./results --max-downloads 10`; `gvf-run` does not expose that limit
291: - Check `--verbose` output for progress
